function num(v) { if (v == null || v === '') return null; const n = Number(v); return Number.isFinite(n) ? n : null; }
function put(ctx, r, c, v) {
  if (v == null) return;
  try { const f = ctx.cell.getFormula(SHEET, r, c); if (f && String(f).trim() !== '') return; ctx.cell.set(SHEET, r, c, v); }
  catch (e) { /* bỏ qua */ }
}

return defineScript({
  async afterInject(ctx) {
    const a = (ctx.shared && ctx.shared.a != null) ? Number(ctx.shared.a)
      : ((ctx.shared && ctx.shared.year) ? Number(ctx.shared.year) : 2025);
    const b = (ctx.shared && ctx.shared.b != null) ? Number(ctx.shared.b) : (a + 1);
    const scope = (ctx.shared && ctx.shared.scopeKind) ? ('&scopeKind=' + ctx.shared.scopeKind) : '';
    const src = (ctx.shared && ctx.shared.sourceSheet) ? ('&sourceSheetCode=' + encodeURIComponent(ctx.shared.sourceSheet)) : '';
    // FORM CHUNG NĂM HỌC: cột "6 tháng cuối" (H/I) = Kỳ 1 = HK1 (T7–12 năm Y);
    //                     cột "6 tháng đầu"  (F/G) = Kỳ 2 = HK2 (T1–6 năm Y+1).
    const HN = [
      { from: a + '-01-01', to: a + '-06-30', qCol: 5, kpCol: 6 }, // 6th đầu F/G ← HK2
      { from: a + '-07-01', to: a + '-12-31', qCol: 7, kpCol: 8 },             // 6th cuối H/I ← HK1
    ];
    let total = 0;
    for (const k of HN) {
      const path = '/rest/qt/summary/data?from=' + k.from + '&to=' + k.to + scope + src;
      let rows;
      try { rows = await ctx.api.get(path); }
      catch (e) { ctx.ui.message('Lỗi lấy số 1m nửa năm ' + k.from + ': ' + e.message); continue; }
      for (const r of (rows || [])) {
        const row = ROW[r[CK]];
        if (row == null) continue;
        put(ctx, row - 1, k.qCol, num(r.quantity));
        put(ctx, row - 1, k.kpCol, num(r.amount));
        total += 1;
      }
    }
    ctx.ui.message('Đã đổ ' + total + ' ô (đối tượng+kinh phí, 2 nửa năm) vào 1m. Cột tổng + tab 01 tự tính.');
  },
});
