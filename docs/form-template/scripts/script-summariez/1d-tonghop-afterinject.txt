// Nguon: DB qt_template_script — bieu tong hop nam (YEAR_SUMMARY), script "1d"
// Hook: afterInject. Dong bo tu ban DA KIEM THU ngay 2026-08-22 (nam duong lich).
// SUA 2026-09-29: chi lay don vi DUOC GIAO bieu trong nam (period-organizations?from&to&reportType)
//   thay cho /rest/qt/master/organization-units (toan bo danh muc don vi).

// ============================================================================
//  SCRIPT TỔNG HỢP · BIỂU 1d (1d-sgd-xp)  —  hook afterInject  (NHÓM B: khối tỉnh/xã · 2 KỲ)
//  Hỗ trợ học sinh khuyết tật. 1 chỉ tiêu sổ cái: 1D.HS_KHUYET_TAT.SO_DOI_TUONG (qty = số ĐT, amt = TỔNG KP).
//  ‼ MISMATCH DỮ LIỆU: sheet tách KP thành Học bổng (F/K) + Mua sắm đồ dùng (G/L), nhưng sổ cái biểu con
//    CHỈ có 1 TỔNG KP (không tách) → script DỒN tổng KP vào cột HỌC BỔNG (F kỳ I / K kỳ II); cột đồ dùng
//    G/L để trống (nhập tay nếu cần tách). E/J = SUM(học bổng, đồ dùng) → tổng vẫn ĐÚNG.
//  ‼ THỨ TỰ KỲ (theo model năm học chuẩn, như 1e): C–G = Kỳ I (HK1 = năm a, T7–12) ; H–L = Kỳ II
//    (HK2 = năm b, T1–6). LƯU Ý: nhãn cột Kỳ II trong template hiện ghi "năm học 2024-2025" (CHƯA cập nhật
//    như 1e = "2025-2026"); nếu 1d dùng đúng năm-học-tiến thì sửa nhãn ô H5 cho khớp.
//  ĐỔ per đơn vị: C/H số ĐT ; D/I số tháng (từ dòng mẫu) ; F/K = tổng KP (dồn học bổng). KP tổng E/J/M
//    + tiểu tổng r10/r13 (SUM range động) + tổng r9 GIỮ CÔNG THỨC template — script KHÔNG đụng.
//  2 khối: type==TINH|SO (tỉnh lên đầu) → Khối tỉnh (mẫu r11) ; MỌI type khác → Khối xã (mẫu r14). Mỗi đơn vị 1 dòng (clone).
//  ĐƠN VỊ: chỉ đơn vị ĐƯỢC GIAO biểu 1d trong năm (kể cả chưa nộp) — /summary/period-organizations?reportType=1d.
// ============================================================================

const SHEET = '1d-sgd-xp';
const CK = 'indi' + 'catorCode';
const MA = '1D.HS_KHUYET_TAT.SO_DOI_TUONG';

// Cột 0-based: Kỳ I  C=2 số ĐT, D=3 số tháng, E=4 KP tổng(ct), F=5 học bổng, G=6 đồ dùng ;
//              Kỳ II H=7 số ĐT, I=8 số tháng, J=9 KP tổng(ct), K=10 học bổng, L=11 đồ dùng ; M=12 tổng(ct).
const B = 1, C_DT_I = 2, C_THANG_I = 3, C_HB_I = 5, C_DT_II = 7, C_THANG_II = 8, C_HB_II = 10;
// Dòng 0-based: r9 tổng=8 ; r10 khối tỉnh=9 ; r11 Sở mẫu=10 (đệm r12) ; r13 khối xã=12 ; r14 xã mẫu=13 (đệm r15).
const SO_ROW = 10, XA_TOT = 12, XA_ROW = 13;

function num(v) { if (v == null || v === '') return null; const n = Number(v); return Number.isFinite(n) ? n : null; }
function put(ctx, r, c, v) {
  if (v == null) return;
  try { const f = ctx.cell.getFormula(SHEET, r, c); if (f && String(f).trim() !== '') return; ctx.cell.set(SHEET, r, c, v); }
  catch (e) { /* bỏ qua */ }
}
function setText(ctx, r, c, v) { if (v == null || v === '') return; try { ctx.cell.set(SHEET, r, c, v); } catch (e) { /* bỏ qua */ } }

// Đơn vị của tab = đơn vị ĐƯỢC GIAO biểu (theo mã), rồi tới đơn vị có số mà không nằm trong danh sách
// giao (dữ liệu cũ) — tên/loại của nhóm sau tra danh mục. KHÔNG lấy cả danh mục đơn vị làm dòng.
async function unitsOf(apiGet, assigned, maps) {
  const units = [], seen = {}, extra = [];
  for (const u of assigned) if (u && u.code && !seen[u.code]) { seen[u.code] = true; units.push(u); }
  for (const m of maps) for (const code in m) {
    if (code && code !== 'TONG_HOP' && !seen[code]) { seen[code] = true; extra.push(code); }
  }
  if (extra.length === 0) return units;
  const byCode = {}; for (const u of await apiGet('/rest/qt/master/organization-units')) byCode[u.code] = u;
  for (const code of extra.sort()) units.push(byCode[code] || { code: code, name: code, orgType: '' });
  return units;
}

return defineScript({
  async afterInject(ctx) {
    const a = (ctx.shared && ctx.shared.a != null) ? Number(ctx.shared.a)
      : ((ctx.shared && ctx.shared.year) ? Number(ctx.shared.year) : 2025);
    const b = (ctx.shared && ctx.shared.b != null) ? Number(ctx.shared.b) : (a + 1);
    const src = (ctx.shared && ctx.shared.sourceSheet) ? ('&sourceSheetCode=' + encodeURIComponent(ctx.shared.sourceSheet)) : '';
    async function apiGet(path) { try { return (await ctx.api.get(path)) || []; } catch (e) { ctx.ui.message('Lỗi 1d: ' + e.message); return []; } }

    // Đơn vị ĐƯỢC GIAO biểu 1d trong năm (kể cả chưa nộp) — KHÔNG lấy cả danh mục đơn vị: đơn vị
    // không được giao biểu 1d thì không có dòng.
    const assigned = await apiGet('/rest/qt/summary/period-organizations?from=' + a + '-01-01&to=' + a
      + '-12-31&reportType=1d');

    // {q: số ĐT, a: tổng KP} per đơn vị (groupByOrg) trong 1 khoảng.
    async function periodByOrg(from, to) {
      const m = {};
      for (const r of await apiGet('/rest/qt/summary/data?from=' + from + '&to=' + to + '&groupByOrg=true' + src)) {
        if (r[CK] !== MA) continue;
        const code = r.organizationCode || '';
        const e = m[code] || (m[code] = { q: 0, a: 0 });
        e.q += (num(r.quantity) || 0); e.a += (num(r.amount) || 0);
      }
      return m;
    }
    const hk1 = await periodByOrg(a + '-07-01', a + '-12-31');   // Kỳ I  (HK1 năm a) → C–G
    const hk2 = await periodByOrg(a + '-01-01', a + '-06-30');   // Kỳ II (HK2 năm b) → H–L

    const units = await unitsOf(apiGet, assigned, [hk1, hk2]);
    const nameOf = {}; for (const u of units) nameOf[u.code] = u.name || u.code;
    // Khối tỉnh = TỈNH + SỞ (tỉnh lên đầu, rồi đến sở theo mã); MỌI loại khác → Khối xã/phường.
    const KHOI_TINH = ['TINH', 'SO'];
    const so = units.filter((u) => KHOI_TINH.includes(u.orgType))
      .sort((x, y) => KHOI_TINH.indexOf(x.orgType) - KHOI_TINH.indexOf(y.orgType));
    const xa = units.filter((u) => !KHOI_TINH.includes(u.orgType));

    // Số tháng đọc từ dòng mẫu (Sở r11 / xã r14) — trước clone.
    const soThangI = num(ctx.cell.get(SHEET, SO_ROW, C_THANG_I)), soThangII = num(ctx.cell.get(SHEET, SO_ROW, C_THANG_II));
    const xaThangI = num(ctx.cell.get(SHEET, XA_ROW, C_THANG_I)), xaThangII = num(ctx.cell.get(SHEET, XA_ROW, C_THANG_II));

    const shift = so.length > 1 ? so.length - 1 : 0;
    if (so.length > 1) ctx.rows.clone(SHEET, SO_ROW, so.length - 1);
    const xaRow = XA_ROW + shift;
    if (xa.length > 1) ctx.rows.clone(SHEET, xaRow, xa.length - 1);

    function writeUnit(r0, u, thangI, thangII) {
      setText(ctx, r0, B, nameOf[u.code] || u.code);
      const e1 = hk1[u.code], e2 = hk2[u.code];
      // Kỳ I: số ĐT + số tháng + KP dồn học bổng (đồ dùng G trống).
      if (e1) { put(ctx, r0, C_DT_I, e1.q); setText(ctx, r0, C_THANG_I, thangI); put(ctx, r0, C_HB_I, e1.a); }
      // Kỳ II: số ĐT + số tháng + KP dồn học bổng (đồ dùng L trống).
      if (e2) { put(ctx, r0, C_DT_II, e2.q); setText(ctx, r0, C_THANG_II, thangII); put(ctx, r0, C_HB_II, e2.a); }
    }
    for (let i = 0; i < so.length; i += 1) writeUnit(SO_ROW + i, so[i], soThangI, soThangII);
    for (let j = 0; j < xa.length; j += 1) writeUnit(xaRow + j, xa[j], xaThangI, xaThangII);

    // KP tổng E/J/M + tiểu tổng r10/r13 + tổng r9: GIỮ CÔNG THỨC template (SUM range động) — script KHÔNG ghi.
    ctx.ui.message('Đã đổ 1d: ' + so.length + ' Sở + ' + xa.length + ' đơn vị khối xã. KP dồn cột Học bổng; tổng tự tính (công thức).');
  },
});
