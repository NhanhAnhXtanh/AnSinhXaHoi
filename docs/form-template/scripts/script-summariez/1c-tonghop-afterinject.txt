// Nguon: DB qt_template_script — bieu tong hop nam (YEAR_SUMMARY), script "1c"
// Hook: afterInject. Dong bo tu ban DA KIEM THU ngay 2026-08-22 (nam duong lich).
// SUA 2026-09-29: chi lay don vi DUOC GIAO bieu trong nam (period-organizations?from&to&reportType)
//   thay cho /rest/qt/master/organization-units (toan bo danh muc don vi).

// ============================================================================
//  SCRIPT TỔNG HỢP · BIỂU 1c (1c-sgd-xp)  —  hook afterInject  (MẪU NHÓM B: khối tỉnh/xã)
//  Chế độ giáo viên mầm non. 1 chỉ tiêu SỐ GIÁO VIÊN (1C.GV_MN.SO_GV). GỘP CẢ NĂM HỌC vào 1 cột.
//  ‼ KỲ theo NĂM HỌC (vắt 2 năm): Kỳ 1 (HK1) = T7–12 của năm y ; Kỳ 2 (HK2) = T1–6 của năm y+1
//  (vd year=2025 → HK1 T7–12/2025 + HK2 T1–6/2026). BỎ hè. Số GV = MAX(HK1, HK2) per đơn vị (cùng GV).
//  2 khối: type==TINH|SO (tỉnh lên đầu) → Khối tỉnh (mẫu r11); MỌI type khác → Khối xã (mẫu r14). Mỗi đơn vị 1 dòng (clone).
//  ĐỔ số GV (C) + set mức D/thời gian E từ DÒNG MẪU (clone không copy value). Kinh phí F (=ROUND(E×C×D))
//  là CÔNG THỨC tự tính. Tiểu tổng r10/r13 (SUM range động) + tổng r9 (SUM) GIỮ CÔNG THỨC template —
//  script KHÔNG đụng (clone tự mở range). Mã ghép qua CK.
//  ĐƠN VỊ: chỉ đơn vị ĐƯỢC GIAO biểu 1c trong năm (kể cả chưa nộp) — /summary/period-organizations?reportType=1c.
// ============================================================================

const SHEET = '1c-sgd-xp';
const CK = 'indi' + 'catorCode';
const MA = '1C.GV_MN.SO_GV';

// Cột 0-based: C=2 số GV, D=3 mức, E=4 thời gian, F=5 kinh phí (công thức).
const C_GV = 2, C_MUC = 3, C_TG = 4, C_KP = 5;
const B = 1;
// Dòng 0-based: r9 tổng=8 ; r10 khối tỉnh=9 ; r11 Sở mẫu=10 ; r13 khối xã=12 ; r14 xã mẫu=13.
const TONG = 8, TINH_TOT = 9, SO_ROW = 10, XA_TOT = 12, XA_ROW = 13;

function num(v) { if (v == null || v === '') return null; const n = Number(v); return Number.isFinite(n) ? n : null; }
function put(ctx, r, c, v) {
  if (v == null) return;
  try { const f = ctx.cell.getFormula(SHEET, r, c); if (f && String(f).trim() !== '') return; ctx.cell.set(SHEET, r, c, v); }
  catch (e) { /* bỏ qua */ }
}
function setText(ctx, r, c, v) { if (v == null || v === '') return; try { ctx.cell.set(SHEET, r, c, v); } catch (e) { /* bỏ qua */ } }

// Đơn vị của tab = đơn vị ĐƯỢC GIAO biểu (theo mã), rồi tới đơn vị có số mà không nằm trong danh sách
// giao (dữ liệu cũ) — tên/loại của nhóm sau tra danh mục. KHÔNG lấy cả danh mục đơn vị làm dòng.
async function unitsOf(apiGet, assigned, maps) {
  const units = [], seen = {}, extra = [];
  for (const u of assigned) if (u && u.code && !seen[u.code]) { seen[u.code] = true; units.push(u); }
  for (const m of maps) for (const code in m) {
    if (code && code !== 'TONG_HOP' && !seen[code]) { seen[code] = true; extra.push(code); }
  }
  if (extra.length === 0) return units;
  const byCode = {}; for (const u of await apiGet('/rest/qt/master/organization-units')) byCode[u.code] = u;
  for (const code of extra.sort()) units.push(byCode[code] || { code: code, name: code, orgType: '' });
  return units;
}

return defineScript({
  async afterInject(ctx) {
    const a = (ctx.shared && ctx.shared.a != null) ? Number(ctx.shared.a)
      : ((ctx.shared && ctx.shared.year) ? Number(ctx.shared.year) : 2025);
    const b = (ctx.shared && ctx.shared.b != null) ? Number(ctx.shared.b) : (a + 1);
    const src = (ctx.shared && ctx.shared.sourceSheet) ? ('&sourceSheetCode=' + encodeURIComponent(ctx.shared.sourceSheet)) : '';
    async function apiGet(path) { try { return (await ctx.api.get(path)) || []; } catch (e) { ctx.ui.message('Lỗi 1c: ' + e.message); return []; } }

    // Đơn vị ĐƯỢC GIAO biểu 1c trong năm (kể cả chưa nộp) — KHÔNG lấy cả danh mục đơn vị: đơn vị
    // không được giao biểu 1c thì không có dòng.
    const assigned = await apiGet('/rest/qt/summary/period-organizations?from=' + a + '-01-01&to=' + a
      + '-12-31&reportType=1c');

    // Số giáo viên per đơn vị: gom 2 kỳ (T1–6 + T7–12) rồi lấy MAX (cùng GV, không cộng đôi).
    async function gvOfPeriod(range) {
      const m = {};
      for (const r of await apiGet('/rest/qt/summary/data?from=' + range.from + '&to=' + range.to + '&groupByOrg=true' + src)) {
        if (r[CK] !== MA) continue;
        m[r.organizationCode] = (num(m[r.organizationCode]) || 0) + (num(r.quantity) || 0);
      }
      return m;
    }
    // NĂM HỌC (vắt 2 năm dương lịch): Kỳ 1 (HK1) = T7–12 của năm y ; Kỳ 2 (HK2) = T1–6 của năm y+1.
    const gvHK1 = await gvOfPeriod({ from: a + '-07-01', to: a + '-12-31' });        // Kỳ 1 = HK1 (năm y)
    const gvHK2 = await gvOfPeriod({ from: a + '-01-01', to: a + '-06-30' }); // Kỳ 2 = HK2 (năm y+1)
    const gvByOrg = {};
    for (const code of new Set([...Object.keys(gvHK1), ...Object.keys(gvHK2)])) {
      gvByOrg[code] = Math.max(num(gvHK1[code]) || 0, num(gvHK2[code]) || 0);   // MAX 2 kỳ (cùng GV)
    }

    const units = await unitsOf(apiGet, assigned, [gvByOrg]);
    const nameOf = {}; for (const u of units) nameOf[u.code] = u.name || u.code;
    // Khối tỉnh = TỈNH + SỞ (tỉnh lên đầu, rồi đến sở theo mã); MỌI loại khác → Khối xã/phường.
    const KHOI_TINH = ['TINH', 'SO'];
    const so = units.filter((u) => KHOI_TINH.includes(u.orgType))
      .sort((x, y) => KHOI_TINH.indexOf(x.orgType) - KHOI_TINH.indexOf(y.orgType));
    const xa = units.filter((u) => !KHOI_TINH.includes(u.orgType));

    // Mức + thời gian đọc từ dòng mẫu (Sở r11 / xã r14) — trước clone.
    const soMuc = num(ctx.cell.get(SHEET, SO_ROW, C_MUC)), soTg = num(ctx.cell.get(SHEET, SO_ROW, C_TG));
    const xaMuc = num(ctx.cell.get(SHEET, XA_ROW, C_MUC)), xaTg = num(ctx.cell.get(SHEET, XA_ROW, C_TG));

    // Clone Khối tỉnh trước (đẩy khối xã xuống), rồi Khối xã.
    const shift = so.length > 1 ? so.length - 1 : 0;
    if (so.length > 1) ctx.rows.clone(SHEET, SO_ROW, so.length - 1);
    const xaRow = XA_ROW + shift, xaTot = XA_TOT + shift;
    if (xa.length > 1) ctx.rows.clone(SHEET, xaRow, xa.length - 1);

    // Đổ 1 đơn vị: số GV + mức/thời gian; kinh phí F công thức tự tính.
    function writeUnit(r0, u, muc, tg) {
      setText(ctx, r0, B, nameOf[u.code] || u.code);
      const gv = num(gvByOrg[u.code]);
      if (gv != null) { put(ctx, r0, C_GV, gv); setText(ctx, r0, C_MUC, muc); setText(ctx, r0, C_TG, tg); }
    }
    for (let i = 0; i < so.length; i += 1) writeUnit(SO_ROW + i, so[i], soMuc, soTg);
    for (let j = 0; j < xa.length; j += 1) writeUnit(xaRow + j, xa[j], xaMuc, xaTg);

    // Tiểu tổng khối (r10/r13) + tổng (r9): GIỮ CÔNG THỨC template (SUM range động) — script KHÔNG ghi.
    // Kinh phí F mỗi dòng (=ROUND(E×C×D)) tự tính; tiểu tổng SUM tự mở khi clone; tổng SUM(C10,C13).
    ctx.ui.message('Đã đổ 1c: ' + so.length + ' Sở + ' + xa.length + ' đơn vị khối xã. Kinh phí + tổng tự tính (công thức).');
  },
});
