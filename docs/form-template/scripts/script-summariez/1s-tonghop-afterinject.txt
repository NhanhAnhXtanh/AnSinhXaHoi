// Nguon: DB qt_template_script — bieu tong hop nam (YEAR_SUMMARY), script "1s"
// Hook: afterInject. Dong bo tu ban DA KIEM THU ngay 2026-08-22 (nam duong lich).

// ============================================================================
//  SCRIPT TỔNG HỢP · BIỂU 1s (1s-snnmt-95xp)  —  hook afterInject  (THEO ĐƠN VỊ động · 1 KỲ NĂM)
//  Hỗ trợ đất trồng lúa. Sở NNMT + 95 xã phường báo cáo — MỖI ĐƠN VỊ 1 DÒNG (danh sách phẳng,
//  KHÔNG khối tỉnh/xã). Clone dòng mẫu r11 cho từng đơn vị CÓ dữ liệu.
//  3 chỉ tiêu DIỆN TÍCH = 3 cột của mỗi dòng: C=2 chuyên lúa · D=3 vùng QH · E=4 đất còn lại.
//  CHỈ đổ diện tích (quantity). Kinh phí F/G/H (=diện tích×mức) + tổng I là CÔNG THỨC → tự tính khi
//  clone dịch. Tổng cộng r10 = SUM(C11:C12) RANGE ĐỘNG: clone chèn dòng vào giữa range → SpreadJS
//  tự mở range đúng số đơn vị (mẫu r11 + đệm r12; bảng phụ r14+ luôn ngoài range). 1 kỳ cả năm.
// ============================================================================

const SHEET = '1s-snnmt-95xp';
const CK = 'indi' + 'catorCode';

const COL = {                        // mã chỉ tiêu -> cột 0-based (diện tích)
  '1S.DAT_CHUYEN_TRONG_LUA': 2,      // C
  '1S.DAT_CHUYEN_VUNG_QH': 3,        // D
  '1S.DAT_TRONG_LUA_CON_LAI': 4,     // E
};
const DATA_ROW = 11;                 // dòng mẫu (1-based) — điểm clone
const B = 1;

function num(v) { if (v == null || v === '') return null; const n = Number(v); return Number.isFinite(n) ? n : null; }
function put(ctx, r, c, v) {
  if (v == null) return;
  try { const f = ctx.cell.getFormula(SHEET, r, c); if (f && String(f).trim() !== '') return; ctx.cell.set(SHEET, r, c, v); }
  catch (e) { /* bỏ qua */ }
}
function setText(ctx, r, c, v) { try { ctx.cell.set(SHEET, r, c, v); } catch (e) { /* bỏ qua */ } }

return defineScript({
  async afterInject(ctx) {
    const a = (ctx.shared && ctx.shared.a != null) ? Number(ctx.shared.a)
      : ((ctx.shared && ctx.shared.year) ? Number(ctx.shared.year) : 2025);
    const b = (ctx.shared && ctx.shared.b != null) ? Number(ctx.shared.b) : (a + 1);
    const src = (ctx.shared && ctx.shared.sourceSheet) ? ('&sourceSheetCode=' + encodeURIComponent(ctx.shared.sourceSheet)) : '';
    async function apiGet(path) { try { return (await ctx.api.get(path)) || []; } catch (e) { ctx.ui.message('Lỗi 1s: ' + e.message); return []; } }

    // Tên đơn vị (mọi loại) — code -> name.
    const units = await apiGet('/rest/qt/master/organization-units');
    const nameOf = {}; for (const u of units) nameOf[u.code] = u.name || u.code;

    // Diện tích per đơn vị per chỉ tiêu (groupByOrg), cả năm.
    const byOrg = {};
    for (const r of await apiGet('/rest/qt/summary/data?from=' + a + '-01-01&to=' + a + '-12-31&groupByOrg=true' + src)) {
      const col = COL[r[CK]]; if (col == null) continue;
      const code = r.organizationCode || '';
      (byOrg[code] || (byOrg[code] = {}))[col] = num(r.quantity);
    }
    const orgs = Object.keys(byOrg);
    if (orgs.length === 0) { ctx.ui.message('1s: chưa có đơn vị nào nộp.'); return; }

    // Clone thêm dòng cho các đơn vị (dòng mẫu r11 giữ nguyên cho đơn vị đầu).
    if (orgs.length > 1) ctx.rows.clone(SHEET, DATA_ROW - 1, orgs.length - 1);

    orgs.forEach((code, i) => {
      const r0 = DATA_ROW - 1 + i;
      setText(ctx, r0, B, nameOf[code] || code);
      const cols = byOrg[code];
      for (const c in cols) put(ctx, r0, Number(c), cols[c]);   // chỉ đổ diện tích
    });

    ctx.ui.message('Đã đổ 1s: ' + orgs.length + ' đơn vị. Kinh phí + tổng + tab 01 tự tính.');
  },
});
