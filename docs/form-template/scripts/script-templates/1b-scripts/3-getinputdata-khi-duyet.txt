// ==== 1b · Script 3/3: GETINPUTDATA — XUẤT SỔ CÁI KHI DUYỆT ====
// displayOrder = 3, enabled = true, gắn biểu reportType = "1b".
// Chạy trên MÁY CHỦ (GraalJS) khi Sở DUYỆT. Chỉ ĐỌC ô, TRẢ VỀ dòng 7 cột.
//
// CƠ CHẾ SỔ CÁI = SỔ ĐỐI CHIẾU: mỗi chỉ tiêu lưu số lượng (qty) + kinh phí (amt) của CHÍNH nó.
//   Ăn trưa: qty=C (đối tượng) + amt=F (kinh phí ăn trưa =C*D*E, công thức).
//   Nấu ăn:  qty=G (định suất) — kinh phí nấu ăn ở chỉ tiêu KINH_PHI riêng (H, nhập tay).
//   GV lớp ghép: amt=I (nhập tay). Mức (D), thời gian (E) khoá → không lưu. Bỏ ô tổng J.

const SHEET = '1b';
const R = 7;

// qty/amt = cột (0-based); null = không có ô tương ứng.
const BINDINGS = [
  { indicatorCode: '1B.AN_TRUA.DOI_TUONG', indicatorName: 'Hỗ trợ tiền ăn trưa › Số đối tượng', qty: 2, amt: 5 },
  { indicatorCode: '1B.NAU_AN.DINH_SUAT', indicatorName: 'Hỗ trợ nấu ăn › Số định suất', qty: 6, amt: null },
  { indicatorCode: '1B.NAU_AN.KINH_PHI', indicatorName: 'Hỗ trợ nấu ăn › Kinh phí', qty: null, amt: 7 },
  { indicatorCode: '1B.GV_LOP_GHEP.KINH_PHI', indicatorName: 'Hỗ trợ giáo viên lớp ghép › Kinh phí', qty: null, amt: 8 },
];
const NOTE_COL = 10;

function num(v) {
  if (v == null || v === '') return null;
  if (typeof v === 'number' && Number.isFinite(v)) return v;
  const n = Number(String(v).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(n) ? n : null;
}
function optionalNote(v) {
  const t = String(v == null ? '' : v).trim();
  return t === '' ? null : t;
}
function cellVal(ctx, col) { return col == null ? null : num(ctx.cell.get(SHEET, R, col)); }

return defineScript({
  getInputData(ctx) {
    const note = optionalNote(ctx.cell.get(SHEET, R, NOTE_COL));
    return BINDINGS
      .map((b, i) => ({
        organizationCode: '',
        organizationName: '',
        indicatorCode: b.indicatorCode,
        indicatorName: b.indicatorName,
        quantity: cellVal(ctx, b.qty),
        amount: cellVal(ctx, b.amt),
        note: i === 0 ? note : null,
      }))
      .filter((r) => (r.quantity != null && r.quantity !== 0)
        || (r.amount != null && r.amount !== 0)
        || r.note != null);
  },
});
