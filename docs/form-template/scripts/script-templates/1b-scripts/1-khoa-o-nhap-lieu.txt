// Hook: onLoad,beforeEdit,beforeCommit,beforePaste. Dong bo tu ban DA KIEM THU ngay 2026-08-22.
// SUA 2026-10-01 (da ghi DB): 3 guard bo qua mode 'summary-edit' — o so cai So do script "So cai So" quyet.

// ==== 1b · Script 1/3: KHÓA/MỞ Ô NHẬP LIỆU ====
// displayOrder = 1, enabled = true, gắn biểu reportType = "1b".
// Nguồn: form-template/1b_DATA.xlsx (sheet '1b', A1:O10). Tọa độ 0-based.
//
// Biểu 1b (Phụ lục 01b — GDMN theo NĐ 105/2020) là BẢNG NGANG một dòng dữ liệu (row 7):
//   C(c2)=Đối tượng ăn trưa | D(c3)=Mức hỗ trợ | E(c4)=Thời gian | F(c5)=Kinh phí =ROUND(C*D*E,0)  ← CÔNG THỨC
//   G(c6)=Số định suất nấu ăn | H(c7)=Kinh phí nấu ăn | I(c8)=Kinh phí GV lớp ghép
//   J(c9)=Tổng =ROUND(F+H+I,0)  ← CÔNG THỨC | K(c10)=Ghi chú
//
// Mức (D) và thời gian (E) là HẰNG SỐ CHÍNH SÁCH → KHOÁ, đơn vị không chỉnh.
// Ô đơn vị NHẬP: C (số đối tượng), G (số định suất), H, I (kinh phí NHẬP TAY, không công thức), K (ghi chú).
// Bỏ F,J (công thức). A1 đơn vị, B3 tháng/năm.

const SHEET = '1b';
const DATA_ROW = 7;
const INPUT_COLS = [2, 6, 7, 8, 10];

function isEditable(row, col) {
  if (row === 0 && col === 0) return true;  // A1 đơn vị (gộp A1:B1)
  if (row === 2 && col === 1) return true;  // B3 tháng/năm
  return row === DATA_ROW && INPUT_COLS.indexOf(col) >= 0;
}

function unlockEditableCells(ctx) {
  ctx.cell.lock(SHEET, 0, 0, false);
  ctx.cell.lock(SHEET, 2, 1, false);
  INPUT_COLS.forEach((col) => ctx.cell.lock(SHEET, DATA_ROW, col, false));
}

function containsOnlyEditableCells(range) {
  if (range.sheetName !== SHEET) return false;
  for (let row = range.row; row < range.row + range.rowCount; row += 1) {
    for (let col = range.col; col < range.col + range.colCount; col += 1) {
      if (!isEditable(row, col)) return false;
    }
  }
  return true;
}

return defineScript({
  onLoad(ctx) {
    if (ctx.mode && ctx.mode !== 'input') return;
    ctx.cell.lockAll(SHEET, true);
    unlockEditableCells(ctx);
    // Tên đơn vị của phiếu (app đưa vào ctx.shared.orgName): ghi vào ô "Chỉ tiêu / Đơn vị"
    // của dòng nhập + dòng ĐƠN VỊ đầu biểu (A1), giống các biểu 1g–1s.
    const _org = ctx.shared && ctx.shared.orgName ? String(ctx.shared.orgName) : '';
    // Tiêu đề (dòng 2) và dòng "Kết quả thực hiện" (B3) theo kỳ của phiếu: mẫu ghi cứng "NĂM 2027" ở tiêu đề và
    // để trống "Tháng …… Năm ……". Chỉ điền khi còn là chữ mẫu — chữ người dùng đã sửa tay được giữ.
    const _year = ctx.shared && ctx.shared.year ? String(ctx.shared.year) : '';
    const _per = ctx.shared && ctx.shared.periodName ? String(ctx.shared.periodName) : '';
    if (_year) {
      const _title = ctx.cell.get(SHEET, 1, 0);
      if (typeof _title === 'string' && /N[ĂA]M\s+\d{4}\s*$/i.test(_title)) {
        ctx.cell.set(SHEET, 1, 0, _title.replace(/(N[ĂA]M)\s+\d{4}\s*$/i, '$1 ' + _year));
      }
    }
    if (_per) {
      const _when = ctx.cell.get(SHEET, 2, 1);
      if (typeof _when === 'string' && _when.indexOf('……') >= 0) ctx.cell.set(SHEET, 2, 1, 'Kết quả thực hiện: ' + _per);
    }
    if (_org) {
      ctx.cell.set(SHEET, DATA_ROW, 1, _org);
      ctx.cell.set(SHEET, 0, 0, 'ĐƠN VỊ (Xã/phường/Sở ngành/Trường): ' + _org);
    }
  },
  // Guard của màn NHẬP: ở sổ cái Sở ('summary-edit') không ý kiến — script 5-so-cai-so quyết ô Sở sửa.
  beforeEdit(ctx, cell) {
    if (ctx.mode === 'summary-edit') return;
    return cell.sheetName === SHEET && isEditable(cell.row, cell.col);
  },
  beforeCommit(ctx, cell) {
    if (ctx.mode === 'summary-edit') return;
    return cell.sheetName === SHEET && isEditable(cell.row, cell.col);
  },
  beforePaste(ctx, range) {
    if (ctx.mode === 'summary-edit') return;
    return containsOnlyEditableCells(range);
  },
});
