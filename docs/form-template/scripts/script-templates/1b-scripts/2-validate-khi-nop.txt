// ==== 1b · Script 2/3: VALIDATE KHI NỘP ====
// displayOrder = 2, enabled = true, gắn biểu reportType = "1b".
// Chạy trong TRÌNH DUYỆT khi bấm NỘP. 'ERROR' chặn nộp; 'WARNING' cảnh báo. Chỉ ĐỌC ô.
// Mức (D) và thời gian (E) KHOÁ cố định → không kiểm. Kiểm các ô đơn vị nhập + ô công thức
// (F,J đọc giá trị đã tính, bắt luôn lỗi công thức). KHÔNG kiểm Ghi chú (K). Dòng = row 7.

const SHEET = '1b';
const DATA_ROW = 7;

const NUMERIC_CELLS = [
  { col: 2, label: 'Số đối tượng ăn trưa' },
  { col: 5, label: 'Kinh phí ăn trưa' },        // F: ô công thức
  { col: 6, label: 'Số định suất nấu ăn' },
  { col: 7, label: 'Kinh phí nấu ăn' },
  { col: 8, label: 'Kinh phí giáo viên lớp ghép' },
  { col: 9, label: 'Tổng kinh phí' },           // J: ô công thức
];

function toNumber(value) {
  if (value == null || value === '') return null;
  if (typeof value === 'number' && Number.isFinite(value)) return value;
  const parsed = Number(String(value).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(parsed) ? parsed : NaN;
}

return defineScript({
  validate(ctx) {
    const issues = [];
    let anyValue = false;

    for (const cell of NUMERIC_CELLS) {
      const n = toNumber(ctx.cell.get(SHEET, DATA_ROW, cell.col));
      if (Number.isNaN(n)) {
        issues.push({ severity: 'ERROR', code: 'NOT_NUMBER', message: `${cell.label}: phải là số.` });
      } else if (typeof n === 'number') {
        if (n < 0) {
          issues.push({ severity: 'ERROR', code: 'NEGATIVE', message: `${cell.label}: không được âm.` });
        }
        if (n > 0) anyValue = true;
      }
    }

    if (!anyValue) {
      issues.push({ severity: 'ERROR', code: 'EMPTY_1B', message: 'Biểu 1b chưa có số liệu nào để nộp.' });
    }
    return issues;
  },
});
