// Hook: beforeEdit,beforeCommit,beforePaste,getSummaryData. Ghi vao DB ngay 2026-10-01 (khong kem 3 dong dau nay).

// Hàm cho SỔ CÁI SỞ — biểu 1b. Nằm CHUNG bộ script của biểu (displayOrder = 5), không còn bộ riêng.
// Hook: beforeEdit, beforeCommit, beforePaste (chỉ có hiệu lực ở mode 'summary-edit'), getSummaryData
// (máy chủ chạy mỗi lần Sở bấm Lưu sổ).
//
// 1b là biểu MỖI ĐƠN VỊ MỘT DÒNG: 4-tonghop-theo-ky để dòng 8 (0-based 7) làm TỔNG CỘNG và thêm
// mỗi đơn vị một dòng bên dưới, tên đơn vị ở cột B.
// - Guard: Sở chỉ sửa ô nhập (không công thức) của các dòng ĐƠN VỊ; dòng Tổng cộng và cột mức / thời
//   gian (hằng số chính sách) khoá.
// - getSummaryData: mỗi dòng đơn vị → các chỉ tiêu có số, organizationCode dịch từ TÊN ở cột B qua
//   ctx.shared.orgCodeByName (máy chủ cấp). Tên không dịch được ⇒ báo lỗi, không đoán.

const SHEET = '1b';
const TOTAL_ROW = 7;
const NAME_COL = 1;
const MAX_UNITS = 2000;
const BINDINGS = [
  { code: '1B.AN_TRUA.DOI_TUONG', name: 'Hỗ trợ tiền ăn trưa › Số đối tượng', qty: 2, amt: 5 },
  { code: '1B.NAU_AN.DINH_SUAT', name: 'Hỗ trợ nấu ăn › Số định suất', qty: 6, amt: null },
  { code: '1B.NAU_AN.KINH_PHI', name: 'Hỗ trợ nấu ăn › Kinh phí', qty: null, amt: 7 },
  { code: '1B.GV_LOP_GHEP.KINH_PHI', name: 'Hỗ trợ giáo viên lớp ghép › Kinh phí', qty: null, amt: 8 },
];
// Ô Sở gõ được: số đối tượng / định suất / kinh phí nhập tay. Cột 5 (kinh phí ăn trưa) là công thức.
const INPUT_COLS = [2, 6, 7, 8];

function num(value) {
  if (value == null || value === '') return null;
  const n = Number(value);
  return Number.isFinite(n) ? n : null;
}

function unitName(ctx, row) {
  const value = ctx.cell.get(SHEET, row, NAME_COL);
  return value == null ? '' : String(value).trim();
}

function editable(ctx, sheetName, row, col) {
  return sheetName === SHEET && row > TOTAL_ROW && INPUT_COLS.indexOf(col) >= 0 && unitName(ctx, row) !== '';
}

return defineScript({
  // Ngoài sổ cái Sở (màn nhập, tổng hợp) không ý kiến — guard của màn nhập ở 1-khoa-o-nhap-lieu.
  beforeEdit(ctx, cell) {
    if (ctx.mode !== 'summary-edit') return;
    return editable(ctx, cell.sheetName, cell.row, cell.col);
  },
  beforeCommit(ctx, cell) {
    if (ctx.mode !== 'summary-edit') return;
    return editable(ctx, cell.sheetName, cell.row, cell.col);
  },
  beforePaste(ctx, range) {
    if (ctx.mode !== 'summary-edit') return;
    for (let row = range.row; row < range.row + range.rowCount; row += 1) {
      for (let col = range.col; col < range.col + range.colCount; col += 1) {
        if (!editable(ctx, range.sheetName, row, col)) return false;
      }
    }
    return true;
  },
  getSummaryData(ctx) {
    const codeByName = (ctx.shared && ctx.shared.orgCodeByName) || {};
    const rows = [];
    for (let i = 0; i < MAX_UNITS; i += 1) {
      const row = TOTAL_ROW + 1 + i;
      const name = unitName(ctx, row);
      if (name === '') break;
      const code = codeByName[name];
      if (!code) {
        ctx.issues.add({ severity: 'ERROR', code: 'UNKNOWN_UNIT',
          message: 'Biểu 1b: không xác định được mã của đơn vị "' + name + '" (dòng ' + (row + 1) + ').' });
        continue;
      }
      for (const b of BINDINGS) {
        const quantity = b.qty == null ? null : num(ctx.cell.get(SHEET, row, b.qty));
        const amount = b.amt == null ? null : num(ctx.cell.get(SHEET, row, b.amt));
        if (quantity == null && amount == null) continue;
        rows.push({ organizationCode: code, indicatorCode: b.code, indicatorName: b.name,
          quantity: quantity, amount: amount, orderNo: i });
      }
    }
    return rows;
  },
});
