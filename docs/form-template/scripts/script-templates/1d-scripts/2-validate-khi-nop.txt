// Hook: validate. Dong bo tu ban DA KIEM THU ngay 2026-08-22.

// ==== 1d · Script 2/3: VALIDATE KHI NỘP ====
// displayOrder = 2, enabled = true, gắn biểu reportType = "1d".
// Chạy trong TRÌNH DUYỆT khi bấm NỘP. 'ERROR' chặn nộp. Chỉ ĐỌC ô.
// Số tháng (D) khoá cố định (=5) → không kiểm. Kiểm Số đối tượng (C) VÀ Tổng kinh phí (H,
// ô công thức — đọc giá trị đã tính). KHÔNG kiểm Ghi chú (I). Dòng dữ liệu = row 7.

const SHEET = '1d';
const DATA_ROW = 7;

const NUMERIC_CELLS = [
  { col: 2, label: 'Số đối tượng' },
  { col: 7, label: 'Tổng kinh phí' },   // H: ô công thức
];

function toNumber(value) {
  if (value == null || value === '') return null;
  if (typeof value === 'number' && Number.isFinite(value)) return value;
  const parsed = Number(String(value).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(parsed) ? parsed : NaN;
}

return defineScript({
  validate(ctx) {
    const issues = [];

    const soDoiTuong = toNumber(ctx.cell.get(SHEET, DATA_ROW, 2));

    for (const cell of NUMERIC_CELLS) {
      const n = toNumber(ctx.cell.get(SHEET, DATA_ROW, cell.col));
      if (Number.isNaN(n)) {
        issues.push({ severity: 'ERROR', code: 'NOT_NUMBER', message: `${cell.label}: phải là số.` });
      } else if (typeof n === 'number' && n < 0) {
        issues.push({ severity: 'ERROR', code: 'NEGATIVE', message: `${cell.label}: không được âm.` });
      }
    }

    if (!(soDoiTuong && soDoiTuong > 0)) {
      issues.push({ severity: 'ERROR', code: 'EMPTY_1D', message: 'Biểu 1d chưa có số đối tượng để nộp.' });
    }
    return issues;
  },
});
