// ==== 1d · Script 3/3: GETINPUTDATA — XUẤT SỔ CÁI KHI DUYỆT ====
// displayOrder = 3, enabled = true, gắn biểu reportType = "1d".
// Chạy trên MÁY CHỦ (GraalJS) khi Sở DUYỆT. Chỉ ĐỌC ô, TRẢ VỀ dòng 7 cột.
//
// CƠ CHẾ SỔ CÁI = SỔ ĐỐI CHIẾU: quantity = Số đối tượng (C, ô nhập); amount = Tổng kinh phí
//   (H, ô công thức — LƯU value đã tính). Số tháng (D=5) + tham số C16:C18 khoá → không lưu.

const SHEET = '1d';
const R = 7;

const BINDINGS = [
  { indicatorCode: '1D.HS_KHUYET_TAT.SO_DOI_TUONG', indicatorName: 'Hỗ trợ học sinh khuyết tật › Số đối tượng', qty: 2, amt: 7 },
];
const NOTE_COL = 8;

function num(v) {
  if (v == null || v === '') return null;
  if (typeof v === 'number' && Number.isFinite(v)) return v;
  const n = Number(String(v).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(n) ? n : null;
}
function optionalNote(v) {
  const t = String(v == null ? '' : v).trim();
  return t === '' ? null : t;
}
function cellVal(ctx, col) { return num(ctx.cell.get(SHEET, R, col)); }

return defineScript({
  getInputData(ctx) {
    const note = optionalNote(ctx.cell.get(SHEET, R, NOTE_COL));
    return BINDINGS
      .map((b, i) => ({
        organizationCode: '',
        organizationName: '',
        indicatorCode: b.indicatorCode,
        indicatorName: b.indicatorName,
        quantity: cellVal(ctx, b.qty),
        amount: cellVal(ctx, b.amt),
        note: i === 0 ? note : null,
      }))
      .filter((r) => (r.quantity != null && r.quantity !== 0)
        || (r.amount != null && r.amount !== 0)
        || r.note != null);
  },
});
