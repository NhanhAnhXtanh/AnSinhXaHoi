// ==== 1s · Script 2/3: VALIDATE KHI NỘP ====
// displayOrder = 2, enabled = true, gắn biểu reportType = "1s".
// Chạy trong TRÌNH DUYỆT khi bấm NỘP. 'ERROR' chặn nộp. Chỉ ĐỌC ô.
// Kiểm 3 diện tích (C,D,E) và Tổng kinh phí (I, ô công thức). KHÔNG kiểm Ghi chú (J). Dòng = row 8.

const SHEET = '1s';
const DATA_ROW = 8;

const NUMERIC_CELLS = [
  { col: 2, label: 'Diện tích đất chuyên trồng lúa' },
  { col: 3, label: 'Trong đó vùng quy hoạch năng suất/chất lượng cao' },
  { col: 4, label: 'Diện tích đất trồng lúa còn lại' },
  { col: 8, label: 'Tổng kinh phí thực hiện' },   // I: ô công thức
];

function toNumber(value) {
  if (value == null || value === '') return null;
  if (typeof value === 'number' && Number.isFinite(value)) return value;
  const parsed = Number(String(value).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(parsed) ? parsed : NaN;
}

return defineScript({
  validate(ctx) {
    const issues = [];
    let anyValue = false;
    for (const cell of NUMERIC_CELLS) {
      const n = toNumber(ctx.cell.get(SHEET, DATA_ROW, cell.col));
      if (Number.isNaN(n)) {
        issues.push({ severity: 'ERROR', code: 'NOT_NUMBER', message: `${cell.label}: phải là số.` });
      } else if (typeof n === 'number') {
        if (n < 0) issues.push({ severity: 'ERROR', code: 'NEGATIVE', message: `${cell.label}: không được âm.` });
        if (n > 0) anyValue = true;
      }
    }
    if (!anyValue) {
      issues.push({ severity: 'ERROR', code: 'EMPTY_1S', message: 'Biểu 1s chưa có số liệu nào để nộp.' });
    }
    return issues;
  },
});
