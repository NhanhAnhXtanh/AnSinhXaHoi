// ==== 1o · Script 3/3: GETINPUTDATA — XUẤT SỔ CÁI KHI DUYỆT ====
// displayOrder = 3, enabled = true, gắn biểu reportType = "1o".
// Chạy trên MÁY CHỦ (GraalJS) khi Sở DUYỆT. Chỉ ĐỌC ô, TRẢ VỀ dòng 7 cột.
//
// CƠ CHẾ SỔ CÁI = SỔ ĐỐI CHIẾU: lưu số hộ (qty) + kinh phí (amt).
//   Hộ nghèo D → kinh phí M (=D*mức*số tháng). Hộ chính sách (F,G) → tổng E=F+G, kinh phí CHUNG N
//   (không tách theo F/G trong form) → lưu ở 1 chỉ tiêu kinh phí riêng 1O.TIEN_DIEN.HO_CS.KINH_PHI.
//   Mức C17 + số tháng C18(=12) khoá → không lưu. Bỏ ô tổng C,E,L.

const SHEET = '1o';
const R = 7;

const BINDINGS = [
  { indicatorCode: '1O.TIEN_DIEN.HO_NGHEO', indicatorName: 'Hỗ trợ tiền điện › Số hộ nghèo', qty: 3, amt: 12 },
  { indicatorCode: '1O.TIEN_DIEN.HO_CS_TCXH_KHAC', indicatorName: 'Hỗ trợ tiền điện › Hộ hưởng trợ cấp xã hội hằng tháng (khác)', qty: 5, amt: null },
  { indicatorCode: '1O.TIEN_DIEN.HO_CS_TCXH_QUY_DINH', indicatorName: 'Hỗ trợ tiền điện › Hộ hưởng trợ cấp xã hội theo quy định', qty: 6, amt: null },
  { indicatorCode: '1O.TIEN_DIEN.HO_CS.KINH_PHI', indicatorName: 'Hỗ trợ tiền điện › Kinh phí hộ chính sách xã hội (chung)', qty: null, amt: 13 },
];
const NOTE_COL = 14;

function num(v) {
  if (v == null || v === '') return null;
  if (typeof v === 'number' && Number.isFinite(v)) return v;
  const n = Number(String(v).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(n) ? n : null;
}
function optionalNote(v) {
  const t = String(v == null ? '' : v).trim();
  return t === '' ? null : t;
}
function cellVal(ctx, col) { return col == null ? null : num(ctx.cell.get(SHEET, R, col)); }

return defineScript({
  getInputData(ctx) {
    const note = optionalNote(ctx.cell.get(SHEET, R, NOTE_COL));
    return BINDINGS
      .map((b, i) => ({
        organizationCode: '',
        organizationName: '',
        indicatorCode: b.indicatorCode,
        indicatorName: b.indicatorName,
        quantity: cellVal(ctx, b.qty),
        amount: cellVal(ctx, b.amt),
        note: i === 0 ? note : null,
      }))
      .filter((r) => (r.quantity != null && r.quantity !== 0)
        || (r.amount != null && r.amount !== 0)
        || r.note != null);
  },
});
