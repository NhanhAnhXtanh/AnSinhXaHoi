// - Guard: Sở chỉ sửa được cột C ở dòng lá (không sửa tham số C33:C37, ghi chú, tiêu đề).
// - getSummaryData: đọc lại cột C (số đối tượng) + cột D (kinh phí đã tính) của từng dòng lá,
//   organizationCode = '' (dòng gộp). Máy chủ so với tổng các đơn vị trong sổ Sở; phần chênh
//   ghi vào đơn vị Sở chọn ở hộp lý do, hoặc dòng "Sở điều chỉnh chung".

const QTY_COL = 2;

const LEAF_ROWS = LEAVES.map((leaf) => leaf.row);

function num(value) {
  if (value == null || value === '') return null;
  const n = Number(value);
  return Number.isFinite(n) ? n : null;
}

function editable(sheetName, row, col) {
  return sheetName === SHEET && col === QTY_COL && LEAF_ROWS.indexOf(row) >= 0;
}

return defineScript({
  // Ngoài sổ cái Sở (màn nhập, tổng hợp) không ý kiến — guard của màn nhập ở 1-khoa-o-nhap-lieu.
  beforeEdit(ctx, cell) {
    if (ctx.mode !== 'summary-edit') return;
    return editable(cell.sheetName, cell.row, cell.col);
  },
  beforeCommit(ctx, cell) {
    if (ctx.mode !== 'summary-edit') return;
    return editable(cell.sheetName, cell.row, cell.col);
  },
  beforePaste(ctx, range) {
    if (ctx.mode !== 'summary-edit') return;
    for (let row = range.row; row < range.row + range.rowCount; row += 1) {
      for (let col = range.col; col < range.col + range.colCount; col += 1) {
        if (!editable(range.sheetName, row, col)) return false;
      }
    }
    return true;
  },
  getSummaryData(ctx) {
    const rows = [];
    LEAVES.forEach(function (leaf, index) {
      const quantity = num(ctx.cell.get(SHEET, leaf.row, QTY_COL));
      const amount = num(ctx.cell.get(SHEET, leaf.row, AMOUNT_COL));
      if (quantity == null && amount == null) return;
      rows.push({ organizationCode: '', indicatorCode: leaf.code, indicatorName: leaf.name,
        quantity: quantity, amount: amount, orderNo: index });
    });
    return rows;
  },
});
