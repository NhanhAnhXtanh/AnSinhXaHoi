function num(value) {
  if (value == null || value === '') return null;
  if (typeof value === 'number' && Number.isFinite(value)) return value;
  const parsed = Number(String(value).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(parsed) ? parsed : null;
}

function optionalNote(value) {
  const text = String(value == null ? '' : value).trim();
  return text === '' ? null : text;
}

function cellNum(ctx, cell) {
  return num(ctx.cell.get(SHEET, cell.row, cell.col));
}

return defineScript({
  getInputData(ctx) {
    return BINDINGS
      .map((b) => ({
        organizationCode: '',
        organizationName: '',
        indicatorCode: b.indicatorCode,
        indicatorName: b.indicatorName,
        quantity: cellNum(ctx, b.quantity),
        amount: cellNum(ctx, b.amount),
        note: optionalNote(ctx.cell.get(SHEET, b.note.row, b.note.col)),
      }))
      .filter((r) => (r.quantity != null && r.quantity !== 0)
        || (r.amount != null && r.amount !== 0)
        || r.note != null);
  },
});
