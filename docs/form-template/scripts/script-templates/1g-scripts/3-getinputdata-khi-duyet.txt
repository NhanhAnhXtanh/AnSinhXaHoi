const NOTE_COL = 14;

function num(v) {
  if (v == null || v === '') return null;
  if (typeof v === 'number' && Number.isFinite(v)) return v;
  const n = Number(String(v).trim().replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(n) ? n : null;
}
function optionalNote(v) {
  const t = String(v == null ? '' : v).trim();
  return t === '' ? null : t;
}
function cellVal(ctx, row, col) { return col == null ? null : num(ctx.cell.get(SHEET, row, col)); }

return defineScript({
  getInputData(ctx) {
    const noteUsed = {};
    return BINDINGS
      .map((b) => {
        let note = null;
        if (!noteUsed[b.row]) {
          note = optionalNote(ctx.cell.get(SHEET, b.row, NOTE_COL));
          noteUsed[b.row] = true;
        }
        return {
          organizationCode: '',
          organizationName: '',
          indicatorCode: b.indicatorCode,
          indicatorName: b.indicatorName,
          quantity: cellVal(ctx, b.row, b.qty),
          amount: cellVal(ctx, b.row, b.amt),
          note: note,
        };
      })
      .filter((r) => (r.quantity != null && r.quantity !== 0)
        || (r.amount != null && r.amount !== 0)
        || r.note != null);
  },
});
